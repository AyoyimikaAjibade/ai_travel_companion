Project ai_travel_companion {
  database_type: 'PostgreSQL'
  Note: '''
    AI-powered travel planning and booking system.
    Supports both authenticated and anonymous users.
  '''
}

// ============================================
// User Management & Authentication Tables
// ============================================

Table users {
  id uuid [pk, default: `gen_random_uuid()`]
  email varchar(255) [unique, not null]
  username varchar(50) [unique, not null]
  hashed_password varchar(255) [not null]
  first_name varchar(100)
  last_name varchar(100)
  is_active boolean [default: true, not null]
  is_superuser boolean [default: false, not null]
  last_login timestamp
  created_time timestamp [default: `CURRENT_TIMESTAMP`, not null]
  updated_time timestamp [default: `CURRENT_TIMESTAMP`, not null]
  
  Note: 'Core user accounts table. Stores authentication credentials and user profile information.'
}

Table user_preferences {
  id uuid [pk, default: `gen_random_uuid()`]
  user_id uuid [ref: > users.id, not null]
  preference_type varchar(50) [not null]
  value jsonb [not null, default: `'{}'`]
  created_time timestamp [default: `CURRENT_TIMESTAMP`, not null]
  updated_time timestamp [default: `CURRENT_TIMESTAMP`, not null]
  
  indexes {
    (user_id, preference_type) [unique]
  }
  
  Note: 'Flexible key-value store for user preferences (theme, notifications, travel preferences, etc.)'
}

Table token_blacklist {
  id uuid [pk, default: `gen_random_uuid()`]
  token_hash varchar(255) [unique, not null]
  user_id uuid [ref: > users.id, not null]
  expires_at timestamp [not null]
  blacklisted_at timestamp [default: `CURRENT_TIMESTAMP`, not null]
  created_time timestamp [default: `CURRENT_TIMESTAMP`, not null]
  updated_time timestamp [default: `CURRENT_TIMESTAMP`, not null]
  
  indexes {
    token_hash
    user_id
    expires_at
  }
  
  Note: 'JWT token blacklist for logout functionality. Stores SHA256 hash of blacklisted tokens.'
}

// ============================================
// Chat & Conversation Management Tables
// ============================================

Table chats {
  id uuid [pk, default: `gen_random_uuid()`]
  user_id uuid [ref: > users.id, null]
  slot_id varchar(26)
  origin_code varchar(3) [not null]
  origin_name varchar(255) [not null]
  destination_code varchar(3) [not null]
  destination_name varchar(255) [not null]
  destination_city_name varchar(255)
  destination_city_code varchar(3)
  start_date date [not null]
  end_date date [not null]
  adults integer [default: 1, not null]
  kids integer [default: 0]
  budget decimal(10,2)
  hotel_request boolean
  hotel_amenities jsonb
  hotel_rating integer
  car boolean
  attractions jsonb
  status varchar(20) [default: 'draft']
  share_code varchar(255) [unique]
  notes text
  created_time timestamp [default: `CURRENT_TIMESTAMP`, not null]
  updated_time timestamp [default: `CURRENT_TIMESTAMP`, not null]
  
  indexes {
    slot_id
    share_code
    user_id
    created_time
  }
  
  Note: '''
    Travel chat sessions container.
    - user_id is nullable to support anonymous users
    - slot_id links to AI service conversation tracking (ULID format)
    - status: draft, booked, confirmed, cancelled
    - share_code allows sharing chats without authentication
  '''
}

Table chat_messages {
  id uuid [pk, default: `gen_random_uuid()`]
  chat_id uuid [ref: > chats.id, not null]
  role varchar(20) [not null]
  content text [not null]
  slot_id varchar(26)
  message_metadata jsonb
  ai_response_data jsonb
  created_time timestamp [default: `CURRENT_TIMESTAMP`, not null]
  updated_time timestamp [default: `CURRENT_TIMESTAMP`, not null]
  
  indexes {
    chat_id
    slot_id
    created_time
  }
  
  Note: '''
    Individual messages in chat conversations.
    - role: 'user' or 'bot'
    - slot_id: Links to AI service message tracking
    - ai_response_data: Stores structured AI response for bot messages
    - message_metadata: Additional context and metadata
  '''
}

// ============================================
// Travel Plans & Itineraries Tables
// ============================================

Table plans {
  plan_id uuid [pk, default: `gen_random_uuid()`]
  chat_id uuid [ref: > chats.id, not null]
  total_price decimal(10,2) [not null]
  score decimal(5,2)
  explanation text
  flight jsonb
  hotel jsonb
  car jsonb
  attractions jsonb
  deeplinks jsonb [default: `'{}'`]
  ai_generated boolean [default: false, not null]
  manual boolean [default: false, not null]
  created_time timestamp [default: `CURRENT_TIMESTAMP`, not null]
  updated_time timestamp [default: `CURRENT_TIMESTAMP`, not null]
  
  indexes {
    chat_id
    created_time
  }
  
  Note: '''
    Travel plans generated from chat conversations.
    - Can be AI-generated (via Gemini/Amadeus) or manually created
    - Stores flight, hotel, car rental, and attractions as JSON
    - deeplinks: External booking links for each component
    - score: AI-assigned relevance score (0-10)
  '''
}

// ============================================
// Relationship Definitions
// ============================================

Ref: user_preferences.user_id > users.id [delete: cascade]
Ref: token_blacklist.user_id > users.id [delete: cascade]
Ref: chats.user_id > users.id [delete: set null]
Ref: chat_messages.chat_id > chats.id [delete: cascade]
Ref: plans.chat_id > chats.id [delete: cascade]
